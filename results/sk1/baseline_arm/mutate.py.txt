#!/usr/bin/env python3
"""Make a mutant copy of the audited repo copy: mutate.py NAME FILE OLD_FILE NEW_FILE [FILE OLD NEW ...]
OLD_FILE/NEW_FILE hold the exact snippet to replace and its replacement. OLD must occur exactly once."""
import os
import shutil
import sys

SRC = "/home/user/sk1_base_out/work/repo"
DST_ROOT = "/home/user/sk1_base_out/mutants"


def main():
    name, rest = sys.argv[1], sys.argv[2:]
    dst = os.path.join(DST_ROOT, name)
    if os.path.exists(dst):
        shutil.rmtree(dst)
    shutil.copytree(SRC, dst)
    for i in range(0, len(rest), 3):
        rel, old_f, new_f = rest[i:i + 3]
        old, new = open(old_f).read(), open(new_f).read()
        path = os.path.join(dst, rel)
        text = open(path).read()
        n = text.count(old)
        if n != 1:
            sys.exit(f"{name}: snippet occurs {n} times in {rel}")
        with open(path, "w") as fh:
            fh.write(text.replace(old, new))
        print(f"{name}: patched {rel}")
    print(dst)


if __name__ == "__main__":
    main()
